# ==============================================================
# EC2 Deployment Guide - Streamlit Chatbot (AWS Bedrock)
# ==============================================================
# This guide walks through deploying the chatbot app on AWS EC2
# so anyone on the internet can access it via a public URL.
# ==============================================================


# ==============================================================
# STEP 1: Launch an EC2 Instance (AWS Console)
# ==============================================================
# 1. Go to AWS Console → EC2 → Launch Instance
# 2. Name: ""chatbot-server""" (or anything you like)
# 3. AMI: Ubuntu 24.04 LTS (free tier eligible)
# 4. Instance type: t2.micro (free tier)
# 5. Key pair: create or select existing ""mlops-key.pem"".pem file (SAVE IT - needed for SSH)
# 6. Security Group - add these Inbound Rules:
#    - SSH        | Port 22   | Source: My IP       (for terminal access)
#    - Custom TCP | Port 8501 | Source: 0.0.0.0/0   (for Streamlit access)
# 7. Click Launch Instance


# ==============================================================
# STEP 2: Connect to EC2 via SSH (run from your local PowerShell)
# ==============================================================
ssh -i "mlops-key.pem" ubuntu@<your-ec2-public-ip>  

# Example:
# ssh -i "mlops-key.pem" ubuntu@100.53.169.74

# Note: make sure you run this from the folder where your .pem file is located
# Or provide the full path: ssh -i "E:\keys\mlops-key.pem" ubuntu@<ip>


# ==============================================================
# STEP 3: Set Up the Server Environment (run on EC2)
# ==============================================================
# Update the system packages
sudo apt update && sudo apt upgrade -y

# Install Python virtual environment support
sudo apt install python3-venv python3-pip -y

# Create the project folder
mkdir ~/chatbot
cd ~/chatbot


# ==============================================================
# STEP 4: Upload Project Files from Local to EC2
# (run from your LOCAL PowerShell, not EC2)
# ==============================================================
scp -i "mlops-key.pem" "E:\GenAi-Project-Udemy\Code_15042025\Code_15042025\chatbot_backend.py" "E:\GenAi-Project-Udemy\Code_15042025\Code_15042025\chatbot_frontend.py" ubuntu@100.53.169.74:~/chatbot/

# Verify files were uploaded (run on EC2):
ls ~/chatbot/
# Should show: chatbot_backend.py  chatbot_frontend.py


# ==============================================================
# STEP 5: Create & Activate Virtual Environment (run on EC2)
# ==============================================================
cd ~/chatbot

# Create virtual environment
python3 -m venv .venv

# Activate it (prompt changes to (.venv))
source .venv/bin/activate


# ==============================================================
# STEP 6: Install Required Packages (run on EC2)
# ==============================================================
pip install --upgrade pip
pip install boto3 langchain langchain-aws langchain-community streamlit PyYAML

python -c "import boto3; import streamlit; import yaml; from langchain_aws import ChatBedrock; from langchain_community.chat_message_histories import ChatMessageHistory; print('All imports successful')"

pip freeze

pip freeze > requirements.txt

cat requirements.txt

pip list

find ~ -name "requirements.txt"

# ==============================================================
# STEP 7: Configure AWS Credentials on EC2
# ==============================================================

# --- Option A: IAM Role (Recommended - no keys needed) ---
# 1. Go to AWS Console → EC2 → Your Instance
# 2. Actions → Security → Modify IAM Role
# 3. Create/attach a role with these policies:
#    - AmazonBedrockFullAccess
# 4. No aws configure needed - EC2 will use the role automatically


# --- Option B: aws configure (Quick setup) ---
AWS_ACCESS_KEY_ID            : <YOUR_AWS_ACCESS_KEY_ID>
AWS_SECRET_ACCESS_KEY        : <YOUR_AWS_SECRET_ACCESS_KEY>
AWS_REGION                   : us-east-1           us-east-1 → US East (N. Virginia)
Default output format [json] : json

aws configure
# AWS Access Key ID     : <your-access-key>
# AWS Secret Access Key : <your-secret-key>
# Default region        : us-east-1
# Default output format : json

# Run this command to verify who you are logged in as??
aws sts get-caller-identity

# The Resource Test (List S3 Buckets)
aws s3 ls

# This shows your default profile — name, key, region, etc
aws configure list

# To see all profiles you have configured:
aws configure list-profiles

# To verify the default profile has the right credentials and region:
aws configure list --profile default

# And to confirm it can actually connect to AWS:
aws sts get-caller-identity --profile default


# ==============================================================
# STEP 8: Run the Streamlit App (run on EC2)
# ==============================================================
# Make sure venv is activated first: source .venv/bin/activate

# Run the app (accessible from the internet)
streamlit run chatbot_frontend.py --server.port 8501 --server.address 0.0.0.0

# Access in browser:
# http://<your-ec2-public-ip>:8501
# Example: http://100.53.169.74:8501


# ==============================================================
# STEP 9: Keep App Running After SSH Session Closes
# ==============================================================
# Use nohup to run it in the background
nohup streamlit run chatbot_frontend.py --server.port 8501 --server.address 0.0.0.0 &

# Check if it is still running: / 
ps aux | grep streamlit

# Find the Process ID
ps -ef | grep streamlit

sudo ss -tulnp | grep 8501

cat nohup.out

# Stop the app:
pkill -f streamlit

kill -9 <process_id>

pkill -f chatbot_frontend.py

# View live logs:
tail -f nohup.out

# MAINTENANCE & TROUBLESHOOTING

cd ~/chatbot
source .venv/bin/activate
